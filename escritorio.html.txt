    <script>
        document.getElementById('nome-escritorio').innerText = localStorage.getItem('gvq_romaneio_logado');
        function sair() { localStorage.clear(); window.location.href = 'index.html'; }

        // APROVAÇÕES DE USUÁRIOS
        function carregarPendentes() {
            let users = JSON.parse(localStorage.getItem('gvq_romaneio_usuarios')) || [];
            let pendentes = users.filter(u => u.perfil === 'escritorio' && !u.aprovado);
            let secao = document.getElementById('secao-aprovacoes');
            let tbodyPend = document.getElementById('lista-pendentes');
            tbodyPend.innerHTML = '';

            if (localStorage.getItem('gvq_romaneio_login') === 'matheus.silveira' && pendentes.length > 0) {
                secao.style.display = 'block';
                pendentes.forEach(p => {
                    tbodyPend.innerHTML += `<tr><td>${p.nome}</td><td>${p.login}</td>
                    <td><button class="btn-aprovar" onclick="mudarStatus('${p.login}', true)">Aprovar</button>
                    <button class="btn-negar" onclick="mudarStatus('${p.login}', false)">Negar</button></td></tr>`;
                });
            } else { secao.style.display = 'none'; }
        }

        function mudarStatus(login, acao) {
            let users = JSON.parse(localStorage.getItem('gvq_romaneio_usuarios')) || [];
            if(acao) { users = users.map(u => u.login === login ? {...u, aprovado: true} : u); alert('Aprovado!'); } 
            else { users = users.filter(u => u.login !== login); alert('Removido!'); }
            localStorage.setItem('gvq_romaneio_usuarios', JSON.stringify(users));
            carregarPendentes();
        }

        // TABELA DE ROMANEIOS
        const romaneios = JSON.parse(localStorage.getItem('gvq_romaneios_salvos')) || [];

        function renderizarTabela() {
            const tbody = document.getElementById('corpo-tabela');
            tbody.innerHTML = '';
            
            if(romaneios.length === 0) {
                tbody.innerHTML = '<tr><td colspan="9" style="text-align:center; padding: 30px; color:#999;">Nenhum romaneio recebido.</td></tr>';
                return;
            }

            romaneios.slice().reverse().forEach((r, index) => {
                let idReal = romaneios.length - 1 - index;
                tbody.innerHTML += `
                    <tr>
                        <td><strong>${r.oc}</strong></td>
                        <td>${r.data}</td>
                        <td>${r.comprador}</td>
                        <td>${r.pecuarista}</td>
                        <td>${r.categoria} (${r.agio_txt})</td>
                        <td><strong style="color:var(--verde-gvq);">${r.cabecas}</strong></td>
                        <td>${r.kg_total}</td>
                        <td><strong style="color:var(--azul-gvq);">${r.valor_final}</strong></td>
                        <td><button class="btn-detalhes" onclick="abrirModal(${idReal})">Ver Fatiamento</button></td>
                    </tr>
                `;
            });
        }

        function abrirModal(index) {
            let r = romaneios[index];
            document.getElementById('m_total').innerText = r.kg_total;
            document.getElementById('m_agio').innerText = r.kg_agio;
            document.getElementById('m_base').innerText = r.kg_base;
            document.getElementById('m_morto').innerText = r.kg_morto;

            let gridPesos = document.getElementById('lista-pesos-modal');
            gridPesos.innerHTML = '';
            r.pesos_individuais.forEach((peso, i) => {
                gridPesos.innerHTML += `<div class="peso-badge"><span style="font-size:10px; color:#666; display:block;">Boi ${i+1}</span>${peso} kg</div>`;
            });

            document.getElementById('modal-detalhes').style.display = 'flex';
        }
        function fecharModal() { document.getElementById('modal-detalhes').style.display = 'none'; }

        window.onload = () => { renderizarTabela(); carregarPendentes(); };
    </script>
